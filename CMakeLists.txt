cmake_minimum_required(VERSION 3.22)
project(JTDrumBus VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/CPM.cmake)
CPMAddPackage(NAME JUCE GITHUB_REPOSITORY juce-framework/JUCE GIT_TAG 7.0.9)
juce_add_plugin(JTDrumBus COMPANY_NAME "JT" PLUGIN_MANUFACTURER_CODE Jtjt PLUGIN_CODE JtDb FORMATS VST3 PRODUCT_NAME "JT-Drum-Bus" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(JTDrumBus PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/DSP.h)
target_compile_definitions(JTDrumBus PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(JTDrumBus PRIVATE juce::juce_audio_utils juce::juce_dsp)
